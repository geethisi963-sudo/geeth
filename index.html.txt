<script>
const nav = document.getElementById('nav');

function showTab(id) {
  document.querySelectorAll('.section').forEach(section => {
    section.classList.toggle('active', section.id === id);
  });

  document.querySelectorAll('nav button').forEach(button => {
    button.classList.toggle('active', button.dataset.tab === id);
  });

  document.getElementById('home').style.display = 'none';
  nav.classList.remove('open');
  window.scrollTo({ top: 0, behavior: 'smooth' });
}

function showHome() {
  document.querySelectorAll('.section').forEach(section =>
    section.classList.remove('active')
  );
  document.getElementById('home').style.display = 'block';
  nav.classList.remove('open');
  window.scrollTo({ top: 0, behavior: 'smooth' });
}

function toggleMenu() {
  nav.classList.toggle('open');
}

function filterEmulators() {
  const query = document.getElementById('emulatorSearch').value.toLowerCase();
  let visible = 0;

  document.querySelectorAll('.emulator').forEach(card => {
    const match = (card.innerText + ' ' + card.dataset.name).toLowerCase().includes(query);
    card.style.display = match ? '' : 'none';
    if (match) visible++;
  });

  document.getElementById('noResults').style.display = visible ? 'none' : 'block';
}

function updateProgress() {
  const boxes = [...document.querySelectorAll('.steps input')];
  const completed = boxes.filter(box => box.checked).length;
  document.getElementById('progressBar').style.width =
    (completed / boxes.length * 100) + '%';
  document.getElementById('progressText').textContent =
    `${completed} of ${boxes.length} tasks completed`;
}

function resetChecklist() {
  document.querySelectorAll('.steps input').forEach(box => box.checked = false);
  updateProgress();
  notify('Checklist reset successfully.');
}

function notify(message) {
  const toast = document.getElementById('toast');
  toast.textContent = message;
  toast.style.display = 'block';
  clearTimeout(window.toastTimer);
  window.toastTimer = setTimeout(() => toast.style.display = 'none', 3500);
}

function buyPackage(packageName) {
  // Replace this with your real WhatsApp number.
  const whatsappNumber = '947XXXXXXXX';
  const message = `Hi! I am interested in the ${packageName} package.`;
  const url = `https://wa.me/${whatsappNumber}?text=${encodeURIComponent(message)}`;

  if (whatsappNumber.includes('X')) {
    notify('Set your WhatsApp number in index.html to activate this button.');
    return;
  }
  window.open(url, '_blank', 'noopener,noreferrer');
}

document.getElementById('year').textContent = new Date().getFullYear();
updateProgress();
